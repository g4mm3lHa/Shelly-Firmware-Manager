#include <tunables/global>

# AppArmor profile of the Shelly Firmware Manager.
#
# Loaded by the Supervisor when the host supports AppArmor; the profile name is
# replaced by the app's full slug when it is loaded. Only what the app needs:
# - run the Python interpreter with the app (/app) and the bundled packages (/deps),
# - read the system libraries, certificates and time zone data of the image,
# - read and write its private storage (/data) – nothing else is writable,
# - TCP/UDP networking (Supervisor, Shellys, shelly.cloud, its two web servers).
# No capabilities, no raw sockets, no mounts, no ptrace, no other programs.
# The container init (docker-init, init: true in config.yaml) starts Python and
# forwards the stop signal of the Supervisor.

profile shelly_firmware_manager flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Signals: stop by the Supervisor/Docker, init <-> Python
  signal (receive) peer=unconfined,
  signal (send,receive) peer=@{profile_name},

  # Networking: IPv4/IPv6 TCP and UDP (DNS), local sockets
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix,
  network netlink raw,

  # Programs: the container init and the Python interpreter only
  /sbin/docker-init ix,
  /dev/init ix,
  /usr/local/bin/python3* ix,

  # Image content: read only (shared libraries and Python extensions need "m")
  / r,
  /app/ r,
  /app/** r,
  /deps/ r,
  /deps/** mr,
  /usr/local/lib/** mr,
  /usr/lib/** mr,
  /lib/** mr,
  /etc/** r,
  /usr/share/zoneinfo/** r,

  # Process information read by Python and the C library
  @{PROC}/** r,
  /sys/devices/system/cpu/** r,
  /sys/fs/cgroup/** r,

  # Private storage of the app: state, firmware, passwords, logs, changelogs
  /data/ r,
  /data/** rwk,

  # Temporary files of the Python standard library
  /tmp/ r,
  /tmp/** rwk,
}
